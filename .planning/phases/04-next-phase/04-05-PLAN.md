### Task 1: Install pip-audit


### Task 2: Create vulnerability scan script

<action>Create `scripts/scan-vulnerabilities.sh` for CI usage.</action>

```bash
#!/bin/bash
set -e

echo "=== mmcli Dependency Vulnerability Scan ==="

# Check if pyproject.toml exists
if [ ! -f pyproject.toml ]; then
    echo "ERROR: pyproject.toml not found"
    exit 1
fi

# Install dependencies in a temporary environment
echo "Setting up virtual environment..."
python3 -m venv /tmp/vuln-scan-env
source /tmp/vuln-scan-env/bin/activate

# Install project dependencies
pip install -e .

# Run vulnerability scan
echo "Running vulnerability scan..."
pip-audit --strict || {
    echo "Vulnerability scan found issues!"
    exit 1
}

echo "No critical vulnerabilities found."
```

```bash
# Make script executable
chmod +x scripts/scan-vulnerabilities.sh
```

<verify>Script runs: `./scripts/scan-vulnerabilities.sh`</verify>

<done>Vulnerability scanning script is functional.</done>

### Task 3: Add to CI workflow

<action>Add vulnerability scan to GitHub Actions workflow.</action>

```yaml
# .github/workflows/security.yml (new file)
name: Security Scan

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]

jobs:
  vulnerability-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'
          
      - name: Install dependencies
        run: |
          pip install pip-audit
      
      - name: Run vulnerability scan
        run: |
          python -m venv .venv
          source .venv/bin/activate
          pip install -e .
          pip-audit --strict
```

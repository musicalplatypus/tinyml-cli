## Objective

Improve input validation patterns to prevent edge cases and DoS attacks.

## Background

**Context:** Current input validation has basic sanitization but lacks:
- Length limits on all inputs
- Character class restrictions
- Component-level validation for paths
- Consistent error handling

## Analysis of Current Implementation

### Existing `_sanitize_input()` Function

```python
def _sanitize_input(input_str: str) -> str:
    """Sanitize input to prevent command injection."""
    sanitized = re.sub(r'[^\w\-./_ ]', '', input_str)
    while '..' in sanitized:
        sanitized = sanitized.replace('..', '')
    sanitized = sanitized.replace('`', '').replace('$', '').replace(';', '')
    if len(sanitized) > 1024:
        sanitized = sanitized[:1024]
    return sanitized
```

**Issues:**
- No length check on original input (DoS vector)
- No character class whitelist (allows unexpected chars)
- No component-level path validation

```python
# mmcli/cli.py (improved _sanitize_input)
def _sanitize_input(input_str: str, max_length: int = 512) -> str:
    """
    Sanitize input to prevent command injection and buffer overflow.
    
    Args:
        input_str: The raw input string
        max_length: Maximum length of sanitized output (default 512)
        
    Returns:
        Sanitized input string
        
    Raises:
        ValueError: If input is too long or contains invalid characters
    """
    # Early length check to prevent DoS
    if len(input_str) > 1024:
        raise ValueError("Input exceeds maximum length of 1024 characters")
    
    # Character class whitelist: alphanumeric, space, and safe symbols
    valid_chars = set('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789 -_./')
    sanitized = ''.join(c for c in input_str if c in valid_chars)
    
    # Remove consecutive dots (path traversal attempt)
    while '..' in sanitized:
        sanitized = sanitized.replace('..', '.')
    
    # Ensure length limit is enforced
    if len(sanitized) > max_length:
        sanitized = sanitized[:max_length]
        
    return sanitized.strip()
```

```python
# mmcli/cli.py (improved _is_safe_path)
def _is_safe_path(path: str) -> bool:
    """
    Check if path is safe for project operations.
    
    Security properties:
    - Blocks absolute paths (prevents reading system files)
    - Blocks path traversal attempts
    - Limits component length to prevent DoS
    """
    # Normalize the path
    try:
        normalized = os.path.normpath(path)
    except Exception:
        return False
    
    # Block empty paths
    if not normalized or normalized == '.':
        return False
    
    # Block absolute paths (only relative paths allowed)
    if normalized.startswith(('/', '\\', '../')):
        return False
    
    # Check for path traversal patterns
    parts = normalized.replace('\\', '/').split('/')
    for part in parts:
        if part == '..':
            return False
        # Limit component length to prevent DoS
        if len(part) > 64:
            return False
    
    # Allow paths under current directory
    try:
        abs_path = os.path.abspath(normalized)
        base_dir = os.path.abspath('.')
        
        if not abs_path.startswith(base_dir + os.sep) and abs_path != base_dir:
            return False
    except Exception:
        return False
    
    return True
```

```python
# tests/test_input_validation.py (new file)
import pytest
from mmcli.cli import _sanitize_input, _is_safe_path


class TestSanitizeInput:
    """Tests for improved input sanitization."""

    def test_length_limit_on_original(self):
        """Original input length is checked."""
        with pytest.raises(ValueError):
            _sanitize_input("a" * 2048)

    def test_component_length_limit(self):
        """Path components are limited to 64 chars."""
        # This is tested via _is_safe_path
        long_component = "x" * 100
        assert not _is_safe_path(f"./{long_component}/file.txt")

    @pytest.mark.parametrize("malformed", [
        "; command",
        "$VAR",
        "`command`",
        "$(command)",
        "| pipe",
        "& background",
    ])
    def test_special_chars_blocked(self, malformed):
        """Malicious characters are removed."""
        result = _sanitize_input(malformed)
        assert ';' not in result
        assert '$' not in result
        assert '`' not in result
        assert '|' not in result

    def test_valid_input_preserved(self):
        """Valid inputs pass through unchanged."""
        valid = "my-project_name/file.txt"
        assert _sanitize_input(valid) == valid


class TestPathValidation:
    """Tests for improved path validation."""

    def test_absolute_paths_blocked(self):
        """Absolute paths should be rejected."""
        assert not _is_safe_path("/etc/passwd")
        assert not _is_safe_path("/var/log/syslog")

    def test_traversal_blocked(self):
        """Path traversal attempts blocked."""
        assert not _is_safe_path("../../etc/passwd")
        assert not _is_safe_path("../../../etc/passwd")

    def test_relative_paths_allowed(self):
        """Valid relative paths are allowed."""
        assert _is_safe_path("./project/data")
        assert _is_safe_path("../other-project")
```

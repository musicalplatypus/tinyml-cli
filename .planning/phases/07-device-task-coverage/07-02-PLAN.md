```python
"""Tests for mmcli device list and task-type coverage completeness."""
import subprocess
import sys
import pytest
from mmcli.cli import TARGET_DEVICES, TASK_TYPES_TIMESERIES, TASK_TYPES_VISION, TASK_TYPES_AUDIO


class TestDeviceCoverage:
    def test_f28e12_in_target_devices(self):
        assert "F28E12" in TARGET_DEVICES

    def test_f28e12_accepted_by_argparse(self):
        """argparse must not reject F28E12 as an unknown device."""
        result = subprocess.run(
            [sys.executable, "-m", "mmcli", "train", "-d", "F28E12", "--help"],
            capture_output=True, text=True,
        )
        # --help always exits 0; what matters is it doesn't error about the device
        assert result.returncode == 0
        assert "F28E12" in result.stdout

    def test_no_duplicates_in_target_devices(self):
        assert len(TARGET_DEVICES) == len(set(TARGET_DEVICES))

    def test_all_c2000_devices_present(self):
        """Core C2000 device set must be complete."""
        c2000_expected = {
            "F280013", "F280015", "F28003", "F28004", "F2837",
            "F28E12", "F28P55", "F28P65", "F29H85", "F29P58", "F29P32",
        }
        assert c2000_expected.issubset(set(TARGET_DEVICES))


class TestAudioTaskDiscoverability:
    def test_task_types_audio_exists(self):
        assert isinstance(TASK_TYPES_AUDIO, list)
        assert len(TASK_TYPES_AUDIO) > 0

    def test_audio_classification_in_task_types_audio(self):
        assert "audio_classification" in TASK_TYPES_AUDIO

    def test_audio_classification_in_train_help(self):
        result = subprocess.run(
            [sys.executable, "-m", "mmcli", "train", "--help"],
            capture_output=True, text=True,
        )
        assert result.returncode == 0
        assert "audio_classification" in result.stdout

    def test_audio_classification_in_run_help(self):
        result = subprocess.run(
            [sys.executable, "-m", "mmcli", "run", "--help"],
            capture_output=True, text=True,
        )
        assert result.returncode == 0
        assert "audio_classification" in result.stdout

    def test_audio_module_accepted(self):
        """mmcli info -m audio must not error on unknown module."""
        result = subprocess.run(
            [sys.executable, "-m", "mmcli", "info", "-m", "audio", "--help"],
            capture_output=True, text=True,
        )
        assert result.returncode == 0


class TestTaskTypeSetsExclusive:
    """Verify task sets are disjoint — no task type lives in two buckets."""
    def test_timeseries_vision_disjoint(self):
        assert not set(TASK_TYPES_TIMESERIES) & set(TASK_TYPES_VISION)

    def test_timeseries_audio_disjoint(self):
        assert not set(TASK_TYPES_TIMESERIES) & set(TASK_TYPES_AUDIO)

    def test_vision_audio_disjoint(self):
        assert not set(TASK_TYPES_VISION) & set(TASK_TYPES_AUDIO)
```

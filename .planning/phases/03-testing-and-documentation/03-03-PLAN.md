### Task 1: Test builder.py

<action>Create `tests/test_config_builder.py` with comprehensive tests.</action>

```python
# tests/test_config_builder.py (new file)
import pytest
from mmcli.builder import build_config, validate_config, write_yaml

def test_build_config_valid_parameters():
    """Build config with valid parameters."""
    config = build_config(
        project_path="/tmp/project",
        model="generic_timeseries_classification",
        device="F28P55"
    )
    assert config is not None
    assert config['project']['path'] == "/tmp/project"

def test_build_config_missing_required():
    """Build config raises error for missing required params."""
    with pytest.raises(ValueError):
        build_config(project_path="/tmp/project")

def test_validate_config_valid():
    """Validate returns True for valid config."""
    config = {"project": {"path": "/tmp"}, "model": "test"}
    assert validate_config(config) is True

def test_validate_config_invalid():
    """Validate returns False for invalid config."""
    config = {"invalid": "structure"}
    assert validate_config(config) is False
```

<verify>Run: `pytest tests/test_config_builder.py -v`</verify>

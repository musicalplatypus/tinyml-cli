import pytest
from mmcli.cli import _is_safe_path, sanitize_path

def test_windows_path_handling(is_windows):
    """Windows paths should be handled correctly."""
    # Windows absolute path
    assert _is_safe_path("C:\\Users\\test\\project") == True
    
    # Windows relative path
    assert _is_safe_path(".\\project\\data") == True
    
    # Windows path traversal should still be blocked
    assert _is_safe_path("..\\..\\windows\\system32") == False

def test_unix_path_handling(is_linux):
    """Unix paths should work as expected."""
    assert _is_safe_path("/tmp/project") == True
    assert _is_safe_path("./project/data") == True
    assert _is_safe_path("../../etc/passwd") == False

def test_macos_temp_paths(is_macos):
    """macOS temp paths should be allowed."""
    assert _is_safe_path("/private/var/folders/xxx/project") == True
    assert _is_safe_path("/var/folders/xxx/project") == True
```

### Task 3: Add path separator normalization tests

def test_path_separator_normalization():
    """Paths should be normalized regardless of separators."""
    # Mixed separators should be normalized
    normalized = sanitize_path("folder\\\\subfolder/../file.txt")
    assert "/" in normalized or "\\" in normalized
    
    # Traversal should be resolved
    assert ".." not in sanitize_path("./folder/../../data")

def test_platform_path_join(is_windows):
    """Path joining should use correct separator."""
    from mmcli.cli import join_paths
    
    result = join_paths("folder", "subfolder", "file.txt")
    if sys.platform == "win32":
        assert "\\" in result
    else:
        assert "/" in result
```

### Task 2: Create full workflow test

<action>Create `tests/test_workflow.py` with integration tests for complete workflows.</action>

**Note (cross-AI review):** mmcli uses argparse, not Click. There is no `click.testing.CliRunner` or exported `cli` object. All tests must invoke the CLI via `subprocess.run([sys.executable, "-m", "mmcli", ...])`.

```python
# tests/test_workflow.py (new file)
import sys
import subprocess
import pytest

def _run_mmcli(*args, **kwargs):
    """Invoke mmcli via subprocess — the only correct invocation pattern (argparse, not Click)."""
    return subprocess.run(
        [sys.executable, "-m", "mmcli", *args],
        capture_output=True, text=True, **kwargs
    )

def test_init_to_analyze(mock_workflow_env, test_workflow_project):
    """Test complete workflow: init → analyze."""
    result = _run_mmcli("analyze", "-i", str(test_workflow_project))
    assert result.returncode == 0
    assert "classes" in result.stdout or "files" in result.stdout

def test_full_pipeline(mock_workflow_env, test_workflow_project):
    """Test init → analyze → recommend pipeline."""
    analyze_result = _run_mmcli("analyze", "-i", str(test_workflow_project))
    assert analyze_result.returncode == 0

    rec_result = _run_mmcli(
        "recommend", "-i", str(test_workflow_project), "-m", "timeseries"
    )
    assert rec_result.returncode == 0
    assert "generic_timeseries_classification" in rec_result.stdout

def test_dry_run_with_mocked_env(mock_workflow_env, test_workflow_project):
    """Test dry-run with mocked environment."""
    result = _run_mmcli(
        "train", "--dry-run", "-i", str(test_workflow_project), "-m", "timeseries"
    )
    assert result.returncode == 0
```

### Task 3: Add regression tests

<action>Add regression tests to prevent workflow breaking changes.</action>

```python
# tests/test_workflow_regression.py (new file)
import sys
import subprocess
import pytest

def _run_mmcli(*args, **kwargs):
    return subprocess.run(
        [sys.executable, "-m", "mmcli", *args],
        capture_output=True, text=True, **kwargs
    )

def test_analyze_preserves_data(mock_workflow_env, test_workflow_project):
    """Analyze should not modify source data."""
    result1 = _run_mmcli("analyze", "-i", str(test_workflow_project))
    result2 = _run_mmcli("analyze", "-i", str(test_workflow_project))
    assert result1.returncode == result2.returncode
```

### Task 2: Create full workflow test

<action>Create `tests/test_workflow.py` with integration tests for complete workflows.</action>

```python
# tests/test_workflow.py (new file)
import pytest
from click.testing import CliRunner
from mmcli.cli import cli

def test_init_to_analyze(mock_workflow_env, test_workflow_project):
    """Test complete workflow: init → analyze."""
    runner = CliRunner()
    
    # Run analyze on the project
    result = runner.invoke(cli, [
        "analyze",
        "-i", str(test_workflow_project)
    ])
    
    assert result.exit_code == 0
    assert "classes" in result.stdout or "files" in result.stdout

def test_full_pipeline(mock_workflow_env, test_workflow_project):
    """Test init → analyze → recommend pipeline."""
    runner = CliRunner()
    
    # Analyze
    analyze_result = runner.invoke(cli, ["analyze", "-i", str(test_workflow_project)])
    assert analyze_result.exit_code == 0
    
    # Recommend (should work with mocked env)
    rec_result = runner.invoke(cli, [
        "recommend",
        "-i", str(test_workflow_project),
        "-m", "timeseries"
    ])
    
    assert rec_result.exit_code == 0
    assert "generic_timeseries_classification" in rec_result.stdout

def test_dry_run_with_mocked_env(mock_workflow_env, test_workflow_project):
    """Test dry-run with mocked environment."""
    runner = CliRunner()
    
    result = runner.invoke(cli, [
        "train",
        "--dry-run",
        "-i", str(test_workflow_project),
        "-m", "timeseries"
    ])
    
    assert result.exit_code == 0
```

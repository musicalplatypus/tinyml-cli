## Objective

Add an interactive shell mode for REPL-like usage of mmcli commands.

## Background

**Context:** Users who run multiple related commands benefit from a persistent session where they can chain operations without re-typing common arguments.

**Current State:**
```bash
# Repeated typing required
mmcli info -m timeseries
mmcli recommend -m timeseries --project ./my-project
mmcli diagnose --full
```

```python
# mmcli/interactive.py (new file)
"""Interactive shell for mmcli."""

import cmd
from typing import List, Optional


class MMCLIShell(cmd.Cmd):
    """Interactive shell for mmcli commands."""
    
    prompt = "mmcli> "
    intro = (
        "Welcome to mmcli interactive shell!\n"
        "Type 'help' for a list of commands.\n"
        "Type 'exit' or press Ctrl+D to quit.\n"
    )
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.current_project = None
        self.current_module = None
    
    # Command methods must follow cmd.Cmd convention: do_<command>
    
    def do_help(self, arg):
        """Show help for commands."""
        if arg:
            # Show help for specific command
            super().do_help(arg)
        else:
            # Custom help with more context
            print("mmcli Interactive Shell Commands:")
            print("")
            print("  info [options]    - Show available models and devices")
            print("  recommend [opts]  - Recommend models for your project")
            print("  analyze [opts]    - Analyze a dataset")
            print("  diagnose [opts]   - Run diagnostics")
            print("  use <project>     - Set current project directory")
            print("  module <name>     - Set default module type")
            print("  exit              - Exit the shell")
            print("")
    
    def do_exit(self, arg):
        """Exit the interactive shell."""
        print("Goodbye!")
        return True
    
    def do_EOF(self, arg):
        """Handle Ctrl+D (EOF)."""
        print()
        return self.do_exit(arg)
    
    # Command aliases
    do_q = do_exit  # q for quit
    do_quit = do_exit  # quit also works
    
    def do_use(self, arg):
        """Set the current project directory."""
        if not arg:
            print(f"Current project: {self.current_project or '(not set)'}")
            return
        
        import os
        path = arg.strip()
        
        if os.path.isdir(path):
            self.current_project = path
            print(f"Project set to: {path}")
        else:
            print(f"Error: Directory not found: {path}")
    
    def do_module(self, arg):
        """Set the default module type."""
        valid_modules = ["timeseries", "vision", "audio"]
        
        if not arg:
            print(f"Current module: {self.current_module or '(not set)'}")
            return
        
        module = arg.strip().lower()
        if module in valid_modules:
            self.current_module = module
            print(f"Module set to: {module}")
        else:
            print(f"Error: Invalid module. Valid options: {', '.join(valid_modules)}")
    
    def do_info(self, arg):
        """Query info about models and devices."""
        from mmcli.cli import run_mmcli
        
        if self.current_module:
            # Add default module
            full_cmd = f"info -m {self.current_module} {arg}"
        else:
            full_cmd = f"info {arg}"
        
        print(f"Running: mmcli {full_cmd}")
        return_code = run_mmcli(full_cmd.split())
        
        if return_code != 0:
            print(f"Command failed with exit code {return_code}")
    
    def do_recommend(self, arg):
        """Recommend models for the current project."""
        from mmcli.cli import run_mmcli
        
        cmd_parts = ["recommend"]
        
        if self.current_module:
            cmd_parts.append(f"-m {self.current_module}")
        
        if self.current_project:
            cmd_parts.append(f"--project {self.current_project}")
        
        # Add any user-provided arguments
        if arg.strip():
            cmd_parts.append(arg)
        
        full_cmd = " ".join(cmd_parts)
        
        print(f"Running: mmcli {full_cmd}")
        return_code = run_mmcli(full_cmd.split())
        
        if return_code != 0:
            print(f"Command failed with exit code {return_code}")
    
    def do_analyze(self, arg):
        """Analyze a dataset."""
        from mmcli.cli import run_mmcli
        
        cmd_parts = ["analyze"]
        
        if self.current_project:
            cmd_parts.append(f"--project {self.current_project}")
        
        if arg.strip():
            cmd_parts.append(arg)
        
        full_cmd = " ".join(cmd_parts)
        
        print(f"Running: mmcli {full_cmd}")
        return_code = run_mmcli(full_cmd.split())
        
        if return_code != 0:
            print(f"Command failed with exit code {return_code}")
    
    def do_diagnose(self, arg):
        """Run diagnostics."""
        from mmcli.cli import run_mmcli
        
        full_cmd = f"diagnose {arg}".strip()
        
        print(f"Running: mmcli {full_cmd}")
        return_code = run_mmcli(full_cmd.split())
        
        if return_code != 0:
            print(f"Command failed with exit code {return_code}")
    
    def do_clear(self, arg):
        """Clear the terminal screen."""
        import os
        os.system("clear" if os.name == "posix" else "cls")
        print()
    
    def emptyline(self):
        """Handle empty input line."""
        # Don't repeat last command (default behavior)
        pass
    
    def precmd(self, line: str) -> str:
        """Pre-process command before execution."""
        if not line:
            return line
        
        # Parse the command and arguments
        parts = line.split(None, 1)
        cmd_name = parts[0] if parts else ""
        
        # Show help for unknown commands
        valid_commands = ["info", "recommend", "analyze", "diagnose", 
                         "use", "module", "help", "exit", "quit", "clear"]
        
        if cmd_name not in valid_commands:
            print(f"Unknown command: {cmd_name}. Type 'help' for a list of commands.")
            return ""
        
        return line


def start_shell():
    """Start the interactive shell."""
    try:
        import prompt_toolkit
        # Use prompt-toolkit for better readline support if available
        from prompt_toolkit import PromptSession
        from prompt_toolkit.completion import WordCompleter
        
        commands = ["info", "recommend", "analyze", "diagnose", 
                   "use", "module", "help", "exit", "clear"]
        
        session = PromptSession()
        
        print("Welcome to mmcli interactive shell!")
        print("Type 'help' for a list of commands.")
        print("Press Ctrl+C to cancel current input, Ctrl+D to exit.")
        print("")
        
        while True:
            try:
                line = session.prompt("mmcli> ")
                
                if not line.strip():
                    continue
                
                if line.strip().lower() in ["exit", "quit", "q"]:
                    break
                
                # For now, fall back to main CLI
                from mmcli.cli import run_mmcli
                return_code = run_mmcli(line.split())
                
                if return_code != 0:
                    print(f"\nCommand failed with exit code {return_code}")
                    
            except KeyboardInterrupt:
                continue
            except EOFError:
                break
        
        print("Goodbye!")
        
    except ImportError:
        # Fall back to standard cmd module
        shell = MMCLIShell()
        shell.cmdloop()


def start_shell_standard():
    """Start the interactive shell using standard cmd module."""
    shell = MMCLIShell()
    try:
        shell.cmdloop()
    except KeyboardInterrupt:
        print("\nInterrupted.")
    finally:
        print("Goodbye!")
```

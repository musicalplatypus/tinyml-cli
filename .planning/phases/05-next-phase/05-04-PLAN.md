# mmcli/batch.py (new file)
"""Batch processing utilities for handling multiple projects."""

import os
import glob
from typing import List, Callable, Any


def expand_project_paths(patterns: List[str]) -> List[str]:
    """
    Expand glob patterns to list of paths.
    
    Args:
        patterns: List of file/directory patterns (can include globs)
        
    Returns:
        List of expanded paths
    """
    paths = []
    for pattern in patterns:
        # Handle both glob patterns and direct paths
        if '*' in pattern or '?' in pattern:
            matches = glob.glob(pattern, recursive=True)
            paths.extend(matches)
        else:
            paths.append(pattern)
    
    # Remove duplicates while preserving order
    seen = set()
    result = []
    for p in paths:
        if p not in seen:
            seen.add(p)
            result.append(p)
    
    return result


def run_batch(
    command_func: Callable,
    paths: List[str],
    **kwargs
) -> dict:
    """
    Run a command on multiple paths.
    
    Args:
        command_func: Function to call for each path
        paths: List of project/dataset paths
        **kwargs: Additional arguments passed to command_func
        
    Returns:
        Dictionary mapping paths to results/errors
    """
    results = {}
    
    for path in paths:
        try:
            result = command_func(path, **kwargs)
            results[path] = {"success": True, "result": result}
        except Exception as e:
            results[path] = {
                "success": False,
                "error": str(e),
                "traceback": None  # Can add traceback if needed
            }
    
    return results


def format_batch_results(results: dict) -> str:
    """
    Format batch results as human-readable output.
    
    Args:
        results: Dictionary from run_batch()
        
    Returns:
        Formatted string with summary and details
    """
    lines = []
    
    # Summary
    success_count = sum(1 for r in results.values() if r.get("success"))
    fail_count = len(results) - success_count
    
    lines.append("=" * 60)
    lines.append(f"BATCH PROCESSING RESULTS")
    lines.append("=" * 60)
    lines.append("")
    lines.append(f"Total: {len(results)}")
    lines.append(f"Success: {success_count}")
    lines.append(f"Failed: {fail_count}")
    lines.append("")
    
    # Detailed results
    if success_count > 0:
        lines.append("SUCCESSFUL:")
        for path, result in results.items():
            if result.get("success"):
                lines.append(f"  ✓ {path}")
        lines.append("")
    
    if fail_count > 0:
        lines.append("FAILED:")
        for path, result in results.items():
            if not result.get("success"):
                error = result.get("error", "Unknown error")
                lines.append(f"  ✗ {path}: {error[:80]}")
    
    return "\n".join(lines)
```

# mmcli/cli.py (modify handlers)

def handle_train(args) -> int:
    """Handle the train command, including batch mode."""
    from mmcli.batch import expand_project_paths, run_batch
    
    if args.batch:
        # Batch mode: process multiple projects
        paths = expand_project_paths(args.batch)
        
        results = run_batch(train_project, paths)
        print(format_batch_results(results))
        return 0 if all(r["success"] for r in results.values()) else 1
    
    elif args.directory:
        # Directory mode: find and process .yaml files
        import glob
        yaml_files = glob.glob(os.path.join(args.directory, "**/*.yaml"), recursive=True)
        
        results = run_batch(train_project, yaml_files)
        print(format_batch_results(results))
        return 0 if all(r["success"] for r in results.values()) else 1
    
    else:
        # Single project mode (original behavior)
        return train_project(args.project)


def handle_analyze(args) -> int:
    """Handle the analyze command, including batch mode."""
    from mmcli.batch import expand_project_paths, run_batch
    from mmcli.analyze import run_analyze as original_analyze
    
    if args.batch:
        paths = expand_project_paths(args.batch)
        
        results = {}
        for path in paths:
            # Run original analyze on each path
            try:
                result = original_analyze(project_path=path)
                results[path] = {"success": True, "result": result}
            except Exception as e:
                results[path] = {"success": False, "error": str(e)}
        
        print(format_batch_results(results))
        return 0 if all(r["success"] for r in results.values()) else 1
    
    # Single project mode (original behavior)
    return original_analyze(project_path=args.project)


def format_batch_results(results: dict) -> str:
    """Format batch processing results."""
    lines = []
    
    success = sum(1 for r in results.values() if r.get("success"))
    total = len(results)
    
    lines.append("=" * 60)
    lines.append(f"BATCH RESULTS: {success}/{total} successful")
    lines.append("=" * 60)
    lines.append("")
    
    for path, result in results.items():
        status = "✓" if result.get("success") else "✗"
        lines.append(f"{status} {path}")
        if not result.get("success"):
            lines.append(f"   Error: {result.get('error', 'Unknown')}")
    
    return "\n".join(lines)
```

# tests/test_batch.py (new file)
import pytest
from mmcli.batch import expand_project_paths, run_batch


class TestBatchProcessing:
    """Tests for batch processing functionality."""

    def test_expand_glob_patterns(self, tmp_path):
        """Glob patterns should be expanded to paths."""
        # Create test files
        (tmp_path / "project1.yaml").write_text("test")
        (tmp_path / "project2.yaml").write_text("test")
        
        patterns = [str(tmp_path / "*.yaml")]
        paths = expand_project_paths(patterns)
        
        assert len(paths) == 2
        assert any("project1" in p for p in paths)
        assert any("project2" in p for p in paths)

    def test_expand_direct_paths(self, tmp_path):
        """Direct paths should be returned as-is."""
        direct = str(tmp_path / "direct.yaml")
        (tmp_path / "direct.yaml").write_text("test")
        
        paths = expand_project_paths([direct])
        assert paths == [direct]

    def test_run_batch_success(self):
        """Batch should succeed when all operations succeed."""
        results = run_batch(lambda p: p, ["a", "b", "c"])
        
        for path in ["a", "b", "c"]:
            assert results[path]["success"] is True
            assert results[path]["result"] == path

    def test_run_batch_with_errors(self):
        """Batch should report failures individually."""
        def failing_cmd(path):
            if path == "fail":
                raise ValueError("Intentional failure")
            return "ok"
        
        results = run_batch(failing_cmd, ["a", "fail", "b"])
        
        assert results["a"]["success"] is True
        assert results["fail"]["success"] is False
        assert results["b"]["success"] is True


class TestBatchResultsFormatting:
    """Tests for batch result formatting."""

    def test_format_results_success(self):
        """Format should show success summary."""
        results = {
            "a.yaml": {"success": True},
            "b.yaml": {"success": True},
        }
        
        formatted = format_batch_results(results)
        
        assert "2/2 successful" in formatted
        assert "✓ a.yaml" in formatted

    def test_format_results_with_failures(self):
        """Format should show failures with errors."""
        results = {
            "a.yaml": {"success": True},
            "b.yaml": {"success": False, "error": "File not found"},
        }
        
        formatted = format_batch_results(results)
        
        assert "1/2 successful" in formatted
        assert "✗ b.yaml" in formatted
        assert "File not found" in formatted

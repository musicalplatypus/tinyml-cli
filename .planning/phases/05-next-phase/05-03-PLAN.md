```python
# mmcli/compare.py (new file)
"""Model comparison functionality."""

from typing import List, Dict, Any
from mmcli.info import query_registry


def fetch_model_info(module_type: str, models: List[str]) -> Dict[str, Dict]:
    """
    Fetch model information for multiple models.
    
    Args:
        module_type: AI module (timeseries, vision, audio)
        models: List of model identifiers
        
    Returns:
        Dictionary mapping model names to their info
    """
    result = {}
    for model in models:
        try:
            info = query_registry(module_type=module_type, task_type=model)
            result[model] = {
                "name": model,
                **info.get("models", {}).get(model, {})
            }
        except Exception as e:
            result[model] = {"error": str(e)}
    
    return result


def compare_models(
    module_type: str,
    models: List[str],
    device: str = None
) -> Dict[str, Any]:
    """
    Compare multiple models and return comparison data.
    
    Args:
        module_type: AI module (timeseries, vision, audio)
        models: List of model identifiers to compare
        device: Target device (optional, filters results)
        
    Returns:
        Comparison result with metrics for each model
    """
    # Fetch info for all models
    model_info = fetch_model_info(module_type, models)
    
    comparison = {
        "module": module_type,
        "models": {},
        "summary": {}
    }
    
    if device:
        comparison["device"] = device
    
    for name, info in model_info.items():
        if "error" not in info:
            comparison["models"][name] = {
                "accuracy": info.get("accuracy", "N/A"),
                "size_kb": info.get("size_bytes", 0) / 1024,
                "params": info.get("parameters", "N/A"),
                "latency_ms": info.get("inference_latency_ms", "N/A"),
            }
    
    # Calculate summary statistics
    sizes = [m["size_kb"] for m in comparison["models"].values() 
             if isinstance(m["size_kb"], (int, float))]
    
    if sizes:
        comparison["summary"]["smallest_model"] = min(sizes)
        comparison["summary"]["largest_model"] = max(sizes)
        comparison["summary"]["average_size_kb"] = sum(sizes) / len(sizes)
    
    return comparison


def format_comparison(comparison: Dict[str, Any]) -> str:
    """Format comparison result as a human-readable table."""
    lines = []
    
    lines.append("=" * 60)
    lines.append("MODEL COMPARISON")
    lines.append("=" * 60)
    
    if "device" in comparison:
        lines.append(f"Device: {comparison['device']}")
    
    # Summary section
    if "summary" in comparison:
        summary = comparison["summary"]
        lines.append("")
        lines.append("SUMMARY:")
        for key, value in summary.items():
            lines.append(f"  {key}: {value}")
    
    # Per-model details
    if "models" in comparison and comparison["models"]:
        lines.append("")
        lines.append("MODEL DETAILS:")
        
        # Determine columns from first model's keys
        if comparison["models"]:
            sample_model = list(comparison["models"].values())[0]
            columns = ["Model"] + [k for k in sample_model.keys() 
                                   if not isinstance(sample_model[k], (dict, list))]
            
            lines.append("")
            lines.append(" | ".join(f"{c:20}" for c in columns))
            lines.append("-" * 60)
            
            for name, model_data in comparison["models"].items():
                row = [f"{name:20}"]
                for col in columns[1:]:
                    val = str(model_data.get(col, "N/A"))
                    row.append(f"{val[:18]:<18}")
                lines.append(" | ".join(row))
    
    lines.append("")
    return "\n".join(lines)
```

```python
def _add_compare_parser(subparsers) -> None:
    p = subparsers.add_parser(
        "compare",
        help="Compare multiple models side-by-side.",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        description=(
            "Compare model configurations to find the best fit for your needs.\n\n"
            "Examples:\n"
            "  mmcli compare -m timeseries --model1 classification --model2 regression\n"
            "  mmcli compare -m vision --device F28P55 --all-models"
        ),
    )
    
    p.add_argument(
        "-m", "--module",
        required=True,
        choices=["timeseries", "vision", "audio"],
        help="AI module type.",
    )
    
    p.add_argument(
        "--model1",
        type=str,
        required=True,
        metavar="MODEL",
        help="First model to compare (task type).",
    )
    
    p.add_argument(
        "--model2",
        type=str,
        required=True,
        metavar="MODEL",
        help="Second model to compare (task type).",
    )
    
    p.add_argument(
        "-d", "--device",
        type=str,
        default=None,
        metavar="DEVICE",
        help="Target device to filter results.",
    )
    
    p.add_argument(
        "--all-models",
        action="store_true",

    )


def main():
    # ... existing parser setup ...
```

```python
def handle_compare(args) -> int:
    """Handle the compare command."""
    from mmcli.compare import compare_models, format_comparison
    
    if args.all_models:
        # Compare all models
        models = ["classification", "regression", "forecasting", "anomalydetection"]
    else:
        models = [args.model1, args.model2]
    
    comparison = compare_models(
        module_type=args.module,
        models=models,
        device=args.device
    )
    
    print(format_comparison(comparison))
    
    return 0
```

```python
# tests/test_compare.py (new file)
import pytest
from unittest.mock import patch, MagicMock
from mmcli.compare import compare_models, format_comparison


class TestModelComparison:
    """Tests for model comparison functionality."""

    @patch("mmcli.compare.query_registry")
    def test_compare_two_models(self, mock_query):
        """Compare two models should return both in result."""
        mock_query.side_effect = [
            {"models": {"classification": {"accuracy": 0.95}}},
            {"models": {"regression": {"accuracy": 0.89}}},
        ]
        
        comparison = compare_models(
            module_type="timeseries",
            models=["classification", "regression"]
        )
        
        assert "classification" in comparison["models"]
        assert "regression" in comparison["models"]

    def test_format_comparison(self):
        """Format should return human-readable string."""
        data = {
            "module": "timeseries",
            "models": {
                "classification": {"accuracy": 0.95, "size_kb": 128},
                "regression": {"accuracy": 0.89, "size_kb": 96},
            },
            "summary": {"smallest_model": 96}
        }
        
        formatted = format_comparison(data)
        
        assert "MODEL COMPARISON" in formatted
        assert "classification" in formatted
        assert "regression" in formatted

    @patch("mmcli.compare.query_registry")
    def test_compare_with_error_handling(self, mock_query):
        """Compare should handle missing models gracefully."""
        mock_query.side_effect = [
            {"models": {"classification": {"accuracy": 0.95}}},
            Exception("Model not found"),
        ]
        
        comparison = compare_models(
            module_type="timeseries",
            models=["classification", "nonexistent"]
        )
        
        assert "nonexistent" in comparison["models"]
        assert "error" in comparison["models"]["nonexistent"]
```
